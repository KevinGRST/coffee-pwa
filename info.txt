Para el dia lunes traer las imagenes y convertir los icnonos    


Add-Type -AssemblyName System.Drawing
$origen = "$PWD\coffee.jpg"
$imagen = [System.Drawing.Image]::FromFile($origen)
$tamanos = @(72, 96, 128, 144, 152, 192, 384, 512)

foreach ($size in $tamanos) {
    $bmp = New-Object System.Drawing.Bitmap($size, $size)
    $grafico = [System.Drawing.Graphics]::FromImage($bmp)
    # Usar interpolación de alta calidad para que los íconos pequeños no se pixelen
    $grafico.InterpolationMode = [System.Drawing.Drawing2D.InterpolationMode]::HighQualityBicubic
    $grafico.DrawImage($imagen, 0, 0, $size, $size)
    
    $nombre = "$PWD\icon-${size}x${size}.png"
    $bmp.Save($nombre, [System.Drawing.Imaging.ImageFormat]::Png)
    
    $grafico.Dispose()
    $bmp.Dispose()
    Write-Host "Generado: icon-${size}x${size}.png"
}
$imagen.Dispose()
